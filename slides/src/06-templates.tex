\section{Syntax}

\begin{frame}[fragile]{Templates}
\protect\hypertarget{templates-1-of-2}{}
\begin{itemize}
\item
  \emph{Template} --- a single code body for a set of related\\ functions
  (\emph{function template}) and related classes\\ (\emph{class template})
\item
  Syntax:
\begin{lstlisting}
template <typename Type>
declaration;
\end{lstlisting}
  \begin{itemize}
  \item
    \lstinline{typename} denotes that we are creating a formal type parameter
    (\lstinline{class} can also be used).
  \item
    \lstinline{Type}, in this example, is the type ``placeholder'' (because it is listed after
    \lstinline{typename}).
  \item
    \lstinline{declaration} is either a function declaration or a class
    declaration.
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}[fragile]{Templates}
\protect\hypertarget{templates-2-of-2}{}
\begin{itemize}
\item
  \lstinline{typename} in the heading refers to any user-defined\\ type or
  built-in type.
\item
  \lstinline{Type} is a formal parameter to the template
\item
  Just as variables are parameters to functions, data types are
  parameters to templates
\end{itemize}
\end{frame}

\section{Function Templates}

\begin{frame}[fragile]{Function Templates}
\protect\hypertarget{function-templates}{}
\begin{itemize}
\item
  Syntax of the function template:
\begin{lstlisting}
template <typename Type>
function declaration;
\end{lstlisting}
\item
  Type is a formal parameter of the template used to:
  \begin{itemize}
  \item
    Specify the parameter types of the function.
  \item
    Specify the return type of the function.
  \item
    Declare variables within the function.
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}[fragile]{Abbreviated Function Templates}

As of C++20, the \lstinline{auto} keyword can be used as a shorthand for parameter type in a function template declaration.

The following function declarations are identical.

\begin{lstlisting}
template <typename Type>
void fun1(Type val);

template <typename Type>
void fun2(const Type& val);
\end{lstlisting}

\begin{lstlisting}
void fun1(auto val);
void fun2(const auto& val);
\end{lstlisting}
\end{frame}

\begin{frame}{Abbreviated Function Templates}
\texttt{g++} may default to an older C++ standard. If so, add \lstinline[language=Bash, identifierstyle=\color{white}]{-std=c++20} or \lstinline[language=Bash, identifierstyle=\color{white}]{-std=c++23}to your compiler arguments\\ to use template abbreviation.

The auto-grader uses the C++23 when checking your code.
\end{frame}

\section{Code Organization}

\begin{frame}{Source Files for Templated Classes}
\protect\hypertarget{header-file-and-implementation-file-of-a-class-template-1-of-2}{}
\begin{itemize}
\item
  Passing a parameter to a function occurs at \emph{run time}.
\item
  Passing a parameter to a class template occurs at \emph{compile time}.
\item
  Therefore, the implementation file cannot be compiled independently of the client
  code.
  \begin{itemize}
  \item
    Can put class- and function-template definitions
    directly in the client code.
  \item
    Can put class- and function-template definitions
    in the same header file.
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Source Files for Templated Classes}
\protect\hypertarget{header-file-and-implementation-file-of-a-class-template-2-of-2}{}
\begin{itemize}
\item
  Another alternative is to put class definition and\\
  function definitions in separate files.
  \begin{itemize}
  \item
    Include the directive to the implementation file at the end of the header
    file.
  \end{itemize}
\item
  In either case, function definitions and client code are compiled
  together.
\end{itemize}
\end{frame}

\section{Java vs. C++}

\begin{frame}{Type Erasure vs Code Generation}
\begin{itemize}
\item
  Java generics use \emph{type erasure}: at runtime, all\\
  generic types are compiled to Object (or primitive\\ wrappers), and type information is lost.
\item 
  C++ templates are \emph{code generators}:\\
  the compiler creates a new, fully typed version of the function or class for each instantiation.
\item
  C++ templates can lead to code bloat if many different types are used.
\item
  However, it also allows for optimizations that are not possible in Java.
\end{itemize}
\end{frame}

\begin{frame}[fragile]{Type Erasure vs Code Generation}

Java generics example
\begin{lstlisting}[basicstyle=\ttfamily\small]
List<Integer> ints = new ArrayList<>();
List<String> strs = new ArrayList<>();
// At runtime, both are really just List<Object>
\end{lstlisting}

C++ templates example
\begin{lstlisting}[basicstyle=\ttfamily\small]
std::vector<int> ints;
std::vector<std::string> strs;
// At compile time, two classes are generate.
\end{lstlisting}

In C++, \lstinline{vector<int>} and \lstinline{vector<string>} are distinct types at compile time, enabling stronger type safety and optimizations.
\end{frame}

\begin{frame}{Benefits of C++ Templates over Java Generics}
\begin{itemize}
\item C++ templates work directly with the actual type,\\ including primitives, so there's zero runtime\\ overhead.
\item C++ templates can perform compile-time logic (template metaprogramming).
\item C++ allows template specialization: you can write different implementations for specific types.
\item With C++20, templates can specify constraints using concepts (e.g., \lstinline{requires Integral<T>}).
\end{itemize}
\end{frame}

\begin{frame}{Java Generics vs C++ Templates}
  \begin{table}[h!]
\centering\small
\begin{tabular}{lll}
\toprule
\textbf{Feature} & \textbf{Java Generics} & \textbf{C++ Templates} \\
\midrule
Runtime behavior & Type erasure  & Code generation per type \\
Performance      & boxing/unboxing & Zero overhead, fully typed \\
Compile-time power & None & Metaprogramming, \texttt{constexpr} \\
Specialization   & Not supported & Full or partial \\
Constraints      & Interfaces only & C++20 Concepts, etc. \\
\bottomrule
\end{tabular}
\caption{Comparison of Java Generics and C++ Templates}
\end{table}

\section{Random Numbers}
\begin{frame}[fragile]{C++11 Random-Number Generator}
\begin{itemize}
\item
  To use C++11 random-number generator functions,\\
  we use an engine and a distributor.
  \begin{itemize}
  \item
    An engine returns unpredictable (random) bits.
  \item
    A distribution returns random numbers whose likelihoods correspond to
    a specific shape such as a uniform or normal distribution.
  \end{itemize}
\item
  The C++11 standard library provides 25 distribution types in five
  categories.
  \begin{itemize}
  \item
    \lstinline{uniform_int_distribution} and
    \lstinline{uniform_real_distribution} fall in the category of uniform
    distributions.
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}[fragile]{C++11 Random-Number Generator}
\begin{lstlisting}[basicstyle=\ttfamily\small]
#include <random>

int main() {
  std::random_device randDevice;
  std::default_random_engine num{ randDevice() };
  std::uniform_int_distribution<int> dist{1, 6};

  // Generate 10 ints that are between 1 and 6.
  for (int count = 1; count <= 10; ++count) {
    std::cout << dist(num) << ' ';
  }
}
\end{lstlisting}
\end{frame}
