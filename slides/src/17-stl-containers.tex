\section{Introduction}

\begin{frame}{Introduction to the STL}

  The C++ standard defines both the

  \begin{itemize}
  \item
    core language and
  \item
    \emph{Standard Template Library (STL)}.
  \end{itemize}

  The STL includes class templates to process lists, stacks, and queues.
\end{frame}

\begin{frame}{Introduction to the STL}
  In this lecture, you will:
  \begin{itemize}
  \item
    Be introduced the Standard Template Library (STL)
  \item
    Become familiar with the three basic components of the STL:\@ containers, iterators, and algorithms.
  \item
    Become familiar with basic operations on vector objects
  \end{itemize}
\end{frame}

\begin{frame}{Components of the STL}
The STL consists of three categories of components:
\begin{enumerate}
\item
  \emph{Containers} are class templates to manage objects of a given type.
\item
  \emph{Iterators} are used to step through the elements of a container.
\item
  \emph{Algorithms} are used to manipulate data.
\end{enumerate}
\end{frame}

\begin{frame}{Containers Types}
\begin{itemize}[<+->]
\item
  \emph{Sequence Containers} implement data structures \\that can be accessed sequentially.
\item
  \emph{Ordered Associative Containers} implement sorted\\ data structures that can be quickly searched\\ (\(O(\log n)\) complexity).
\item
  \emph{Unordered Associative Containers} implement unsorted (hashed) data structures that can be quickly searched (\(O(1)\) on average, \(O(n)\) worst-case complexity).
\end{itemize}
\end{frame}

\begin{frame}{Container Adapters and Views}

Container adapters and views support different\\
features of existing containers, rather than being\\
containers themselves.

\begin{itemize}[<+->]
\item
  \emph{Adapters} provide a different interface for sequential containers.
\item
  \emph{Views} provide flexible lightweight non-owning \\access to contiguous contains.
\end{itemize}
\end{frame}

\section{Sequence Containers}

\begin{frame}{Sequence Containers in the STL}
Sequence container: every object in the container has a specific position.

\begin{enumerate}
\item
  \lstinline{array} -- encapsulates a fixed size array
\item
  \lstinline{vector} -- a list with elements stored contiguously
\item
  \lstinline{deque} -- a double-ended queue
\item
  \lstinline{list} --  usually a doubly-linked list
  \item
  \lstinline{forward\_list} --  usually a singly-linked list
\end{enumerate}
\end{frame}

\begin{frame}[fragile]{Declaring a \texttt{\bfseries vector}}
\begin{itemize}
\item
  Stores and manages its objects in a dynamic array
\item
  Must use: \lstinline{#include <vector>}
\item
  To define an object of type vector, specify the type of the object
\item
  Examples:
  \begin{lstlisting}
std::vector<int> intList;
std::vector<std::string> stringList;
\end{lstlisting}
\item
  \lstinline{vector} contains several constructors.
\end{itemize}
\end{frame}

\begin{frame}[fragile]{Example use of the \texttt{\bfseries vector} Container}
  \small
\begin{lstlisting}
#include <vector>
#include <iostream>

int main() {
  std::vector<int> v(3);    // a vector of size 3
  v[0] = 23;
  v[1] = 12;
  v[2] = 9;                 // vector full
  v.push_back(17);          // append a new value
  for (auto el : v)         // range-based for loop.
    std::cout << el << ' '; // access to i-th el
  std::cout << std::endl;
  v[2] = v[3];              // random access
}
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]{Example use of the \texttt{\bfseries vector} Container}
  \small
\begin{lstlisting}
#include <vector>
#include <iostream>

int main() {
  std::vector<int> v{ 12, 3, 17, 8 }; // initialize
  while (!v.empty())                  // until empty
  {
      std::cout << v.back() << ' '; // output last el
      v.pop_back();                 // delete last el
  }
  std::cout << std::endl;
}
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]{The \texttt{\bfseries deque} (Double-Ended Queue) Container}
\href{https://en.cppreference.com/w/cpp/container/deque}{\lstinline{std::deque}} is another indexed sequence container, which is \\accessed using
\lstinline{#include <deque>}

\begin{itemize}
\item
  Allows fast insertion and deletion at both end ends. Slower insertion in the middle.
\item
  Insertion and deletion at either end of a deque never invalidates pointers or references to the rest of the elements.
\item
  The elements of a deque are not stored contiguously, but random access is $O(1)$.
\end{itemize}
\end{frame}

\begin{frame}[fragile]{The \texttt{\bfseries list}  Container}
\href{https://en.cppreference.com/w/cpp/container/list}{\lstinline{std::list}} is another indexed sequence container, which is accessed using\\
\lstinline{#include <list>}

\begin{itemize}
\item
  Allows constant-time insertion and deletion from anywhere in the container, which never invalidates pointers or references to the rest of the elements.
\item
  Fast random access is not supported.
\item
  Usually implemented as a doubly-linked list.
\item
  Less space efficient.
\end{itemize}
\end{frame}

\begin{frame}[fragile]{Iterator Example}
  \small
\begin{lstlisting}
template <typename Type>
Type maxValue(const vector<Type> &vec) {
  vector<int>::const_iterator iter = vec.cbegin();
  Type max = *iter;
  for (++iter; iter != vec.cend(); ++iter) {
    if (*iter > max)
      max = *iter;
  }
  return max;
}
\end{lstlisting}

\begin{frame}[fragile]{Types of Iterators}
Every algorithm requires an iterator with a certain\\
capability.
\pause{}
\begin{enumerate}[<+->]
\item
  \emph{Input iterators} have read access; step forward element-by-element.
\item
  \emph{Output iterators} have write access; step forward element-by-element.
\item
  \emph{Forward iterators} have the functionality of input and most of the functionality of output iterators.

\end{enumerate}
\end{frame}

\begin{frame}[fragile]{Types of Iterators}
\begin{enumerate}[<+->]
  \setcounter{enumi}{3}
\item
  \emph{Bidirectional iterators} are forward iterators that\\ can also go backward.
\item
  \emph{Random access iterators} are bidirectional iterators that can randomly process the elements of a container.
\item 
  As of C++20, there are additional iterators based on \href{https://en.cppreference.com/w/cpp/language/constraints}{concepts}.
\end{enumerate}

\pause{}
\href{https://en.cppreference.com/w/cpp/iterator}{https://en.cppreference.com/w/cpp/iterator}
\end{frame}

\begin{frame}{Benefits of Iterators}
\begin{itemize}\setlength\itemsep{0.5em}
\item<2->
  \emph{Convenience in Programming}:
  \begin{itemize}
  \item
    Provide common usage for any container (the internal\\ structure of a container does not matter).
  \item
    Provide read and/or write access to each element's values.
  \end{itemize}
\item<3->
  \emph{Reusable Code}:\\
  Iterator algorithms are not dependent on the container type.
\item<4->
  \emph{Dynamic Processing}:\\
  Easily dynamically allocate as well as delete the memory.
\end{itemize}
\end{frame}

\section{Associative Containers}

\begin{frame}{Ordered Associative Containers}

Store elements in sorted order according to some\\ ordering criteria (often using a type of binary tree).\\[1em]

\begin{tabular}{l l l}
\toprule%
  \textbf{Associative Containers} & \textbf{Header File} & \textbf{Iterator Support}\\
\midrule%
  \href{https://en.cppreference.com/w/cpp/container/map}{map}
    & \lstinline{<map>} & Bidirectional\\
  \href{https://en.cppreference.com/w/cpp/container/multima}{multimap}
    & \lstinline{<map>} & Bidirectional\\
  \href{https://en.cppreference.com/w/cpp/container/set}{set}
    & \lstinline{<set>} & Bidirectional\\
  \href{https://en.cppreference.com/w/cpp/container/multiset}{multiset}
    & \lstinline{<set>} & Bidirectional\\
\bottomrule%
\end{tabular}

\begin{itemize}
\item
  \lstinline{set} stores a set of \emph{unique} elements, while \lstinline{map} stores key-value pairs with \emph{unique} keys.
\item
  \lstinline{multiset} and \lstinline{multimap} allow duplicates.
\end{itemize}

\end{frame}

\begin{frame}[fragile]{Sets Example}
\small
\begin{lstlisting}[basicstyle=\footnotesize\ttfamily]
#include <set>
int main() {
  std::set<std::string, std::less<std::string> > nameSet{ "Ole",
       "Hedvig", "Juan", "Lars", "Guido", "Ann"};

  nameSet.insert("Patric"); // insert another name
  nameSet.erase("Juan"); // removes an element

  // Find the matching name in set.
  std::string searchName;
  std::cin >> searchName;
  std::set<std::string, std::less<string> >::iterator iter =
    nameSet.find(searchName);

  cout << searchName << " is";
  if (iter == nameSet.cend()) // check if found
     std::cout << " NOT";
  std::cout << " in the set!\n";
}
\end{lstlisting}
\end{frame}

    "Maria", "Ann" };

  // Set iterator to lower start value "K"
  std::set<std::string, std::less<std::string> >::iterator
    iter = nameSet.lower_bound("K");

  // Display: Lars, Maria, Ole, Patric
  while (iter != nameSet.upper_bound("Q"))
    cout << *iter++ << ", ";
  std::cout << std::endl;
}
\end{lstlisting}
