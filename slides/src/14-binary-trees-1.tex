\section*{Intro}

\begin{frame}{Motivation: The Problem with Arrays}
  \emph{Binary search} is an efficient search algorithm that relies\\ on splitting a sorted list in half with each comparison.

  Binary search works well on arrays. However,
  \begin{itemize}
    \item 
      Insertion in an array is impossible, requiring the data to be copied to a new array. Deletion is also slow.
    \item
      Finding a specific value in a linked list is slow (needed for searching, insertion, and deletion).
  \end{itemize}
\end{frame}

\begin{frame}{Motivation: The Problem with Linked Lists}
  \emph{Linked Lists}, while great for insertion and deletion at the\\
  ends, pose challenges for binary search due to the lack\\
  of contiguous memory. However,

  \begin{itemize}
    \item 
      Must start iterating from the first/last node (i.e., sequential search).
    \item
      Cannot jump right to the middle to perform a binary search.
  \end{itemize}
\end{frame}


\begin{frame}{Motivation: Solution}
  \textbf{Solution}: Modify the linked-lists structure into a \emph{binary\\
  search tree}, where the binary search can be performed \\
  in logarithmic time (\textit{usually}).

  \pause%
  Binary Search Trees:
  \begin{itemize}
    \item Are \emph{binary trees} that maintain the \emph{binary-search property} (\textit{more on that in a moment}).
    \item Have the benefits of binary search with the flexibility of linked lists.
    \item Are fundamental in computer science and widely used in various algorithms and applications.
  \end{itemize}
\end{frame}

\begin{frame}{Objectives}
In this lecture, we will:

\begin{itemize}
\item
  Explore various binary-tree traversal algorithms.
\item
  Explore how to implement the basic operations on a binary tree.
\item
  Learn about binary \textit{search} trees to:
  \begin{itemize}
    \item
      Organize data
    \item
      Insert and delete items
  \end{itemize}
\item
  Explore recursive and non-recursive tree traversal algorithms.

\end{itemize}
\end{frame}

\section{Binary Search Tree}


\begin{frame}{Binary Search Tree}

\begin{columns}
\column{0.60\textwidth}
  Definition: A \emph{binary search tree} is either empty or has these properties:
  \begin{itemize}[<+->]
    \item
      Has a root node.
    \item
      Has two sets of nodes: the left and right subtrees.
    \item
      The key in the root node is larger than every key in the left subtree and smaller than every key in the right subtree.
    \item
      The left subtree and right subtree are binary search trees.
  \end{itemize}

\begin{frame}{Inserting Values into a Binary Search Tree}
\begin{center}
  \Large
  How do we handle duplicate keys?
\end{center}
\pause{}
We must choose one of three options. The duplicate key is\ldots

\begin{enumerate}
  \item not inserted (don't permit duplicate keys).
  \item inserted to the \emph{left} of the matching key.
  \item inserted to the \emph{right} of the matching key.
\end{enumerate}

As long as the \lstinline{insert()} and \lstinline{contains()} functions follow the same behavior, everything will work.

\section{Traversal}
\begin{frame}{Traversing the Entier Tree}
We must ``\emph{visit}'' all the nodes in a tree determine its size, display it,
delete it, sum the values, etc.\\[1em]
\begin{columns}
\onslide<2->\column{0.3\textwidth}
  In-order
  \begin{enumerate}
  \item
    Traverse the left subtree.
  \item
    \textbf{\color{CSUcyan}Visit the node.}
  \item
    Traverse the right subtree.
  \end{enumerate}
\onslide<3->\column{0.3\textwidth}
  Pre-order
  \begin{enumerate}
  \item
    \textbf{\color{CSUcyan}Visit the node.}
  \item
    Traverse the left subtree.
  \item
    Traverse the right subtree.
  \end{enumerate}
\onslide<4->\column{0.3\textwidth}
  Post-order
  \begin{enumerate}
  \item
    Traverse the left subtree.
  \item
    Traverse the right subtree.
  \item
    \textbf{\color{CSUcyan}Visit the node.}
  \end{enumerate}
\end{columns}
\end{frame}
